local function Create(cls, props)
    local i = Instance.new(cls)
    for k,v in pairs(props) do i[k]=v end
    return i
end

local function Tween(o,t,g,s)
    TweenService:Create(o,TweenInfo.new(t,s or Enum.EasingStyle.Quad,Enum.EasingDirection.Out),g):Play()
end

local function TweenBack(o,t,g)
    TweenService:Create(o,TweenInfo.new(t,Enum.EasingStyle.Back,Enum.EasingDirection.Out),g):Play()
end

local function PlaySound(id,vol)
    local s=Instance.new("Sound",camera); s.SoundId=id; s.Volume=vol or 0.4; s:Play()
    Debris:AddItem(s,3)
end

local SFX_CLICK = "rbxassetid://7145300020"
local SFX_PLACE = "rbxassetid://130501290862644"

local function Snap(v,g) return math.floor(v/g+0.5)*g end
local function Round(v,d) local m=10^(d or 1); return math.floor(v*m+0.5)/m end

local function CamLock(on)
    if on then
        local d=(camera.CFrame.Position-camera.Focus.Position).Magnitude
        player.CameraMaxZoomDistance=math.max(d,0.1)
        player.CameraMinZoomDistance=math.max(d,0.1)
    else
        player.CameraMaxZoomDistance=S.DefaultMaxZoom
        player.CameraMinZoomDistance=S.DefaultMinZoom
    end
end

local function GuiParent()
    local ok=pcall(function() return game:GetService("CoreGui").RobloxGui end)
    return ok and game:GetService("CoreGui") or player:WaitForChild("PlayerGui")
end

local function SafeDestroy(entry)
    if type(entry)=="table" then
        if entry.part and entry.part.Parent then entry.part:Destroy() end
    elseif typeof(entry)=="Instance" and entry.Parent then
        entry:Destroy()
    end
end

-- Highlight helpers
local function HighlightPart(part, col)
    if not part or not part.Parent then return end
    local old=part:FindFirstChild("_SSH"); if old then old:Destroy() end
    Create("SelectionBox",{Name="_SSH",Adornee=part,
        Color3=col or Color3.fromRGB(255,200,0),LineThickness=0.06,Parent=part})
end
local function ClearHighlight(part)
    if part then local h=part:FindFirstChild("_SSH"); if h then h:Destroy() end end
end

local function MakeSlider(page,label,minV,maxV,defV,step,accentCol,onChange)
    step=step or 1; accentCol=accentCol or Color3.fromRGB(0,160,255)
    local frame=Create("Frame",{
        Size=UDim2.new(1,-4,0,50),BackgroundColor3=Color3.fromRGB(20,20,28),
        BorderSizePixel=0,Parent=page})
    Create("UICorner",{CornerRadius=UDim.new(0,6),Parent=frame})
    Create("TextLabel",{
        Size=UDim2.new(0.55,0,0,20),Position=UDim2.new(0,10,0,4),
        BackgroundTransparency=1,Text=label,
        TextColor3=Color3.fromRGB(165,165,185),Font=Enum.Font.Gotham,
        TextSize=12,TextXAlignment=Enum.TextXAlignment.Left,Parent=frame})
    local valLbl=Create("TextLabel",{
        Size=UDim2.new(0.4,0,0,20),Position=UDim2.new(0.55,0,0,4),
        BackgroundTransparency=1,Text=tostring(defV),TextColor3=accentCol,
        Font=Enum.Font.GothamBold,TextSize=12,
        TextXAlignment=Enum.TextXAlignment.Right,Parent=frame})
    Create("UIPadding",{PaddingRight=UDim.new(0,10),Parent=valLbl})
    local track=Create("Frame",{
        Size=UDim2.new(1,-20,0,8),Position=UDim2.new(0,10,0,30),
        BackgroundColor3=Color3.fromRGB(32,32,44),BorderSizePixel=0,Parent=frame})
    Create("UICorner",{CornerRadius=UDim.new(0,4),Parent=track})
    local pct=(defV-minV)/(maxV-minV)
    local fill=Create("Frame",{Size=UDim2.new(pct,0,1,0),BackgroundColor3=accentCol,
        BorderSizePixel=0,Parent=track})
    Create("UICorner",{CornerRadius=UDim.new(0,4),Parent=fill})
    local knob=Create("Frame",{
        Size=UDim2.new(0,16,0,16),AnchorPoint=Vector2.new(0.5,0.5),
        Position=UDim2.new(pct,0,0.5,0),BackgroundColor3=Color3.new(1,1,1),
        BorderSizePixel=0,Parent=track})
    Create("UICorner",{CornerRadius=UDim.new(1,0),Parent=knob})
    Create("UIStroke",{Color=accentCol,Thickness=2,Parent=knob})

    local curVal=defV; local dragging=false
    local function SetFromRel(rel)
        rel=math.clamp(rel,0,1)
        local snapped=math.floor((minV+(maxV-minV)*rel)/step+0.5)*step
        snapped=math.clamp(Round(snapped,3),minV,maxV)
        curVal=snapped
        fill.Size=UDim2.new(rel,0,1,0); knob.Position=UDim2.new(rel,0,0.5,0)
        valLbl.Text=tostring(snapped)
        if onChange then onChange(snapped) end
    end
    track.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 then
            dragging=true
            SetFromRel((i.Position.X-track.AbsolutePosition.X)/track.AbsoluteSize.X)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and i.UserInputType==Enum.UserInputType.MouseMovement then
            SetFromRel((i.Position.X-track.AbsolutePosition.X)/track.AbsoluteSize.X)
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 then dragging=false end
    end)
    return frame, function() return curVal end, function(v)
        SetFromRel(math.clamp((v-minV)/(maxV-minV),0,1))
    end
end

-- Move Part A by delta in world or local space
local function MoveA(axis,delta)
    if not S.SelectedPart or not S.SelectedPart.Parent then return end
    local p=S.SelectedPart
    local offset = axis=="X" and Vector3.new(delta,0,0)
        or axis=="Y" and Vector3.new(0,delta,0)
        or Vector3.new(0,0,delta)
    p.CFrame=p.CFrame+offset
end
local function RotA(axis,deg)
    if not S.SelectedPart or not S.SelectedPart.Parent then return end
    local p=S.SelectedPart
    local r=math.rad(deg)
    local rot = axis=="X" and CFrame.Angles(r,0,0)
        or axis=="Y" and CFrame.Angles(0,r,0)
        or CFrame.Angles(0,0,r)
    p.CFrame=p.CFrame*rot
end
local function ResizeA(axis,delta)
    if not S.SelectedPart or not S.SelectedPart.Parent then return end
    local p=S.SelectedPart
    local s=p.Size
    if axis=="X" then p.Size=Vector3.new(math.max(0.1,s.X+delta),s.Y,s.Z)
    elseif axis=="Y" then p.Size=Vector3.new(s.X,math.max(0.1,s.Y+delta),s.Z)
    else p.Size=Vector3.new(s.X,s.Y,math.max(0.1,s.Z+delta)) end
end

local Ghost=nil
local GhostCols={
    Normal=function() return ActiveColor() end,
    Physics=function() return Color3.fromRGB(255,140,30) end,
    Anchor=function() return Color3.fromRGB(120,120,160) end,
    Script=function() return Color3.fromRGB(20,200,20) end,
    Kill=function() return Color3.fromRGB(255,30,30) end,
    Damage=function() return Color3.fromRGB(255,120,0) end,
    Sensor=function() return Color3.fromRGB(0,160,255) end,
    Moving=function() return ActiveColor() end,
    SpeedPad=function() return Color3.fromRGB(0,255,100) end,
}

local function EnsureGhost()
    if not Ghost or not Ghost.Parent then
        Ghost=Create("Part",{
            Name="SSv9_Ghost",Transparency=0.5,
            CanCollide=false,CanQuery=false,CastShadow=false,
            Anchored=true,Parent=workspace})
        Create("SelectionBox",{Adornee=Ghost,Color3=Color3.new(1,1,1),
            LineThickness=0.04,Parent=Ghost})
    end
end

local function UpdateGhost()
    local wantGhost = (S.Enabled or S.PendingSpecial~=nil) and not toolMode
    if not wantGhost then
        if Ghost then Ghost:Destroy(); Ghost=nil end; return
    end
    EnsureGhost()
    if mouse.Hit then
        local g=S.Grid
        Ghost.CFrame=CFrame.new(
            Snap(mouse.Hit.Position.X,g),
            Snap(mouse.Hit.Position.Y,g)+S.SizeY/2,
            Snap(mouse.Hit.Position.Z,g)
        )*CFrame.Angles(math.rad(S.RotX),math.rad(S.RotY),math.rad(S.RotZ))
        Ghost.Size=Vector3.new(S.SizeX,S.SizeY,S.SizeZ)
        Ghost.Material=S.ActiveMaterial
        Ghost.Shape=S.ActiveShape
        local key=S.PendingSpecial or S.BlockType
        local fn=GhostCols[key]; Ghost.Color=fn and fn() or ActiveColor()
    end
end

-- ═══════════════════════════════════════════════════════════
-- SPAWN SPECIAL BLOCK
-- ═══════════════════════════════════════════════════════════
local function SpawnSpecialAt(pos)
    local type_=S.PendingSpecial
    local g=S.Grid
    local sx,sy,sz=S.SizeX,S.SizeY,S.SizeZ
    local cf=CFrame.new(Snap(pos.X,g),Snap(pos.Y,g)+sy/2,Snap(pos.Z,g))
             *CFrame.Angles(math.rad(S.RotX),math.rad(S.RotY),math.rad(S.RotZ))

    local function Base(col,mat,tr,noColl)
        return Create("Part",{Size=Vector3.new(sx,sy,sz),CFrame=cf,
            Color=col,Material=mat or Enum.Material.Neon,
            Transparency=tr or 0,CanCollide=not noColl,
            Anchored=true,Parent=workspace})
    end
    local function BB(p,txt,col)
        local bg=Create("BillboardGui",{Size=UDim2.new(0,110,0,26),
            StudsOffset=Vector3.new(0,sy/2+1.5,0),AlwaysOnTop=true,Parent=p})
        Create("TextLabel",{Size=UDim2.new(1,0,1,0),BackgroundTransparency=1,
            Text=txt,TextColor3=col,Font=Enum.Font.GothamBold,TextSize=13,Parent=bg})
    end

    local block

    if type_=="Script" then
        block=Base(Color3.fromRGB(20,200,20))
        BB(block,"⌨ SCRIPT",Color3.new(0,1,0))
        local code=S.ScriptCode
        block.Touched:Connect(function(hit)
            if Players:GetPlayerFromCharacter(hit.Parent) then
                local fn=loadstring("local part,hit=...\n"..code)
                if fn then pcall(fn,block,hit) end
            end
        end)

    elseif type_=="Kill" then
        block=Base(Color3.fromRGB(255,30,30))
        BB(block,"☠ KILL",Color3.fromRGB(255,80,80))
        local t=0
        RunService.Heartbeat:Connect(function(dt)
            if not block.Parent then return end
            t=t+dt; block.Color=Color3.new(1,math.abs(math.sin(t*4))*0.2,0)
        end)
        block.Touched:Connect(function(hit)
            local hum=hit.Parent:FindFirstChildOfClass("Humanoid")
            if hum then hum.Health=0 end
        end)

    elseif type_=="Damage" then
        local dmg=S.DamageAmt
        block=Base(Color3.fromRGB(255,120,0))
        BB(block,"⚠ -"..dmg.." HP",Color3.fromRGB(255,180,0))
        local t=0
        RunService.Heartbeat:Connect(function(dt)
            if not block.Parent then return end
            t=t+dt; block.Color=Color3.new(1,0.25+math.abs(math.sin(t*3))*0.35,0)
        end)
        local cd={}
        block.Touched:Connect(function(hit)
            local pl=Players:GetPlayerFromCharacter(hit.Parent)
            if not pl or cd[pl] then return end
            cd[pl]=true
            local hum=hit.Parent:FindFirstChildOfClass("Humanoid")
            if hum then hum.Health=math.max(0,hum.Health-dmg) end
            task.delay(0.8,function() cd[pl]=nil end)
        end)

    elseif type_=="Sensor" then
        block=Base(Color3.fromRGB(0,160,255),Enum.Material.ForceField,0.75,true)
        BB(block,"◉ SENSOR",Color3.fromRGB(0,200,255))
        local inside={}
        block.Touched:Connect(function(hit)
            local pl=Players:GetPlayerFromCharacter(hit.Parent)
            if pl and not inside[pl] then inside[pl]=true
                print("[SSv9 Sensor] "..pl.Name.." ENTERED") end
        end)
        block.TouchEnded:Connect(function(hit)
            local pl=Players:GetPlayerFromCharacter(hit.Parent)
            if pl then inside[pl]=nil; print("[SSv9 Sensor] "..pl.Name.." LEFT") end
        end)

    elseif type_=="Moving" then
        block=Base(ActiveColor(),S.ActiveMaterial,0,false)
        local origin=block.Position; local ax,rng,spd=S.MoveAxis,S.MoveRange,S.MoveSpeed; local t=0
        RunService.Heartbeat:Connect(function(dt)
            if not block.Parent then return end
            t=t+dt; local off=math.sin(t*spd)*rng
            if     ax=="X" then block.CFrame=CFrame.new(origin+Vector3.new(off,0,0))
            elseif ax=="Y" then block.CFrame=CFrame.new(origin+Vector3.new(0,off,0))
            else               block.CFrame=CFrame.new(origin+Vector3.new(0,0,off)) end
        end)

    elseif type_=="SpeedPad" then
        block=Create("Part",{Size=Vector3.new(sx,0.3,sz),CFrame=cf,
            Color=Color3.fromRGB(0,255,100),Material=Enum.Material.Neon,
            Anchored=true,Parent=workspace})
        BB(block,"▶▶ SPEED",Color3.fromRGB(0,255,100))
        block.Touched:Connect(function(hit)
            local root=hit.Parent:FindFirstChild("HumanoidRootPart")
            if root then
                local bv=Create("BodyVelocity",{
                    Velocity=root.CFrame.LookVector*80,
                    MaxForce=Vector3.new(1e5,0,1e5),Parent=root})
                Debris:AddItem(bv,0.3)
            end
        end)
    end

    if block then
        local fs=block.Size; block.Size=Vector3.new(0.1,0.1,0.1); TweenBack(block,0.22,{Size=fs})
        PlaySound(SFX_PLACE,0.4)
        table.insert(S.UndoStack,{part=block})
    end
    DisarmSpecial()
end

-- SINGLE UNIFIED CLICK HANDLER
-- Priority: special spawn > normal place
-- Tool mode selection runs in PARALLEL (doesn't block building)
-- ═══════════════════════════════════════════════════════════
mouse.Button1Down:Connect(function()
    if UIS:GetFocusedTextBox() then return end

    -- Tool mode: select Part A from world — runs regardless of build state
    if toolMode and mouse.Target and mouse.Target~=workspace.Terrain
       and not S.PendingSpecial then
        ClearHighlight(S.SelectedPart)
        S.SelectedPart=mouse.Target
        HighlightPart(S.SelectedPart,Color3.fromRGB(255,200,0))
        selALbl.Text="  A: "..S.SelectedPart.Name
        -- Don't return here — if build is also on, still allow placing
        -- (user is in tool tab clicking world parts, build mode likely off anyway)
        return
    end

    -- Special block armed — place it
    if S.PendingSpecial then
        if mouse.Hit then SpawnSpecialAt(mouse.Hit.Position) end
        return
    end

    -- Normal build
    if S.Enabled then
        PlaceNormal()
    end
end)

        S.ColorG=math.clamp(tonumber(args[3]),0,255)

    elseif args[1]=="clear" then
        for _,e in pairs(S.UndoStack) do SafeDestroy(e) end; S.UndoStack={}
    end
    CmdBar.Text=""
end)

    CT.Active=true; CT.Name=name; CT.Uses=uses; CT.Fn=fn
    CTHUD.Visible=true
    local u = uses==0 and "INF" or tostring(uses)
    CTHUD.Text = "  "..name.."  ["..u.." uses]  — click world to fire  |  Ctrl+Z = cancel"
end

local function DisarmCursor()
    CT.Active=false; CTHUD.Visible=false
end

local function FireCursor(pos)
    if not CT.Active or not CT.Fn then return end
    pcall(CT.Fn, pos)
    if CT.Uses > 0 then
        CT.Uses = CT.Uses - 1
        if CT.Uses == 0 then
            DisarmCursor()
            Notify("Done — "..CT.Name.." used up", Color3.fromRGB(200,200,60))
        else
            CTHUD.Text = "  "..CT.Name.."  ["..CT.Uses.." uses]  — click world to fire  |  Ctrl+Z = cancel"
        end
    end
end

-- ════════════════════════════════════════════════════════════════
-- FLY  (PlatformStand method — only reliable FE approach)
-- ════════════════════════════════════════════════════════════════
local flyBV, flyBG, flyActive = nil, nil, false

local function StartFly()
    if flyActive then return end
    local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    local hum = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
    if not hrp then Notify("No character!"); return end
    flyActive = true
    if hum then hum.PlatformStand = true end
    flyBV = Create("BodyVelocity",{Name="_FlyBV", Velocity=Vector3.new(0,0,0),
        MaxForce=Vector3.new(1e6,1e6,1e6), Parent=hrp})
    flyBG = Create("BodyGyro",{Name="_FlyBG",
        MaxTorque=Vector3.new(4e5,4e5,4e5), D=100, P=8000,
        CFrame=hrp.CFrame, Parent=hrp})
    Notify("FLY ON  WASD+Space/Ctrl  Shift=fast", Color3.fromRGB(80,160,255))
end

local function StopFly()
    flyActive = false
    if flyBV and flyBV.Parent then flyBV:Destroy() end
    if flyBG and flyBG.Parent then flyBG:Destroy() end
    flyBV, flyBG = nil, nil
    local hum = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
    if hum then hum.PlatformStand = false end
    Notify("FLY OFF", Color3.fromRGB(160,160,160))

local function MakeSlider2(pg,lbl,mn,mx,def,step,ac,onChange)
    local fr=Create("Frame",{Size=UDim2.new(1,-4,0,48),
        BackgroundColor3=Color3.fromRGB(20,20,28),BorderSizePixel=0,Parent=pg})
    Create("UICorner",{CornerRadius=UDim.new(0,6),Parent=fr})
    Create("TextLabel",{Size=UDim2.new(0.55,0,0,18),Position=UDim2.new(0,8,0,4),
        BackgroundTransparency=1,Text=lbl,TextColor3=Color3.fromRGB(160,160,180),
        Font=Enum.Font.Gotham,TextSize=12,TextXAlignment=Enum.TextXAlignment.Left,Parent=fr})
    local vl=Create("TextLabel",{Size=UDim2.new(0.4,0,0,18),Position=UDim2.new(0.55,0,0,4),
        BackgroundTransparency=1,Text=tostring(def),TextColor3=ac,
        Font=Enum.Font.GothamBold,TextSize=12,TextXAlignment=Enum.TextXAlignment.Right,Parent=fr})
    Create("UIPadding",{PaddingRight=UDim.new(0,8),Parent=vl})
    local tr=Create("Frame",{Size=UDim2.new(1,-16,0,7),Position=UDim2.new(0,8,0,28),
        BackgroundColor3=Color3.fromRGB(30,30,42),BorderSizePixel=0,Parent=fr})
    Create("UICorner",{CornerRadius=UDim.new(0,3),Parent=tr})
    local pct=(def-mn)/(mx-mn)
    local fi=Create("Frame",{Size=UDim2.new(pct,0,1,0),BackgroundColor3=ac,
        BorderSizePixel=0,Parent=tr})
    Create("UICorner",{CornerRadius=UDim.new(0,3),Parent=fi})
    local kn=Create("Frame",{Size=UDim2.new(0,14,0,14),AnchorPoint=Vector2.new(0.5,0.5),
        Position=UDim2.new(pct,0,0.5,0),BackgroundColor3=Color3.new(1,1,1),
        BorderSizePixel=0,Parent=tr})
    Create("UICorner",{CornerRadius=UDim.new(1,0),Parent=kn})
    Create("UIStroke",{Color=ac,Thickness=2,Parent=kn})
    local cur=def; local drag=false
    local function SFR(rel)
        rel=math.clamp(rel,0,1)
        local sn=math.floor((mn+(mx-mn)*rel)/step+0.5)*step
        sn=math.clamp(sn,mn,mx); cur=sn
        fi.Size=UDim2.new(rel,0,1,0); kn.Position=UDim2.new(rel,0,0.5,0)
        vl.Text=tostring(sn); onChange(sn)
    end
    tr.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 then
            drag=true; SFR((i.Position.X-tr.AbsolutePosition.X)/tr.AbsoluteSize.X) end
    end)
    UIS.InputChanged:Connect(function(i)
        if drag and i.UserInputType==Enum.UserInputType.MouseMovement then
            SFR((i.Position.X-tr.AbsolutePosition.X)/tr.AbsoluteSize.X) end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 then drag=false end
    end)
end

UIS.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    -- F = fly toggle
    if input.KeyCode==Enum.KeyCode.F and S.FlyEnabled then
        if flyActive then StopFly() else StartFly() end
    end
    -- N = noclip toggle
    if input.KeyCode==Enum.KeyCode.N and S.NoclipEnabled then
        S.NoclipActive=not S.NoclipActive
        Notify(S.NoclipActive and "NOCLIP ON" or "NOCLIP OFF",
            S.NoclipActive and Color3.fromRGB(180,80,255) or Color3.fromRGB(160,160,160))
    end
    -- Ctrl+Z also cancels cursor tool (added on top of v9's undo)
    if input.KeyCode==Enum.KeyCode.Z
    and (UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.RightControl)) then
        if CT.Active then DisarmCursor() end
    end
end)

-- ════════════════════════════════════════════════════════════════
-- EXTENDED CLICK HANDLER
-- v9 already has a click handler. We intercept BEFORE it fires
-- by checking cursor tool active state first.
-- We replace mouse.Button1Down entirely here for clarity.
-- ════════════════════════════════════════════════════════════════
-- Disconnect v9's handler by wrapping — actually Roblox fires ALL
-- connected handlers. So we use a guard flag: if cursor tool fires,
-- set a flag so the v9 handler skips its logic.
local _cursorFired = false
mouse.Button1Down:Connect(function()
    if UIS:GetFocusedTextBox() then return end
    if CT.Active and mouse.Hit then
        _cursorFired=true
        FireCursor(mouse.Hit.Position)
        task.delay(0,function() _cursorFired=false end)
    end
end)

-- Patch: wrap the PendingSpecial / build / toolMode check
-- by prepending a check for _cursorFired in the existing v9 handler.
-- Since we can't remove v9's handler, we use S.Enabled guard:
-- cursor tool temporarily disables S.Enabled while firing.
-- Actually simplest: cursor tool check is first in THIS new handler,
-- v9's handler checks S.Enabled which we preserve.
-- The _cursorFired flag prevents double-placing.
-- We also override v9's build click by resetting:
local _origEnabled = false
RunService.Heartbeat:Connect(function()
    -- If cursor tool is active, disable build so v9 handler skips placing
    if CT.Active then
        _origEnabled = S.Enabled
        S.Enabled = false
    elseif _origEnabled and not CT.Active then
        S.Enabled = _origEnabled
        _origEnabled = false
    end
end)

-- ════════════════════════════════════════════════════════════════
-- NOCLIP LOOP  (Stepped = before physics, most reliable)
-- ════════════════════════════════════════════════════════════════
RunService.Stepped:Connect(function()
    if S.NoclipActive then
        local c=GetChar2()
        if c then
            local hum=GetHum2()
            if hum then hum.PlatformStand=true end
            for _,v in pairs(c:GetDescendants()) do
                if v:IsA("BasePart") then v.CanCollide=false end
            end
        end
    else
        if not flyActive then
            local hum=GetHum2()
            if hum and hum.PlatformStand then hum.PlatformStand=false end
        end
    end
end)

-- ════════════════════════════════════════════════════════════════
-- FLY LOOP  (RenderStepped for smooth movement)
-- ════════════════════════════════════════════════════════════════
RunService.RenderStepped:Connect(function(dt)
    if flyActive and flyBV and flyBV.Parent and flyBG and flyBG.Parent then
        local hrp=player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        local hum=GetHum2()
        if hrp then
            if hum then hum.PlatformStand=true end
            local spd = UIS:IsKeyDown(Enum.KeyCode.LeftShift) and 130 or 60
            local vel = Vector3.new(0,0,0)
            local cf  = camera.CFrame
            if UIS:IsKeyDown(Enum.KeyCode.W) then vel=vel+cf.LookVector*spd end
            if UIS:IsKeyDown(Enum.KeyCode.S) then vel=vel-cf.LookVector*spd end
            if UIS:IsKeyDown(Enum.KeyCode.A) then vel=vel-cf.RightVector*spd end
            if UIS:IsKeyDown(Enum.KeyCode.D) then vel=vel+cf.RightVector*spd end
            if UIS:IsKeyDown(Enum.KeyCode.Space)       then vel=vel+Vector3.new(0,60,0) end
            if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then vel=vel-Vector3.new(0,60,0) end
            flyBV.Velocity=vel
            flyBG.CFrame=cf
        end
    end
    -- GOD MODE
    if S.GodEnabled then
        local hum=GetHum2()
        if hum and hum.Health<hum.MaxHealth then hum.Health=hum.MaxHealth end
    end
    -- ESP
    if S.ESPEnabled then
        for _,p2 in pairs(Players:GetPlayers()) do
            if p2~=player and p2.Character then
                local hr=p2.Character:FindFirstChild("HumanoidRootPart")
                if hr and not hr:FindFirstChild("_ESP") then
                    Create("SelectionBox",{Name="_ESP",Adornee=p2.Character,
                        Color3=Color3.fromRGB(255,50,50),LineThickness=0.05,Parent=hr})
                end
            end
        end
    end
end)
